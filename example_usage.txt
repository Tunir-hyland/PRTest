# Example Usage Commands

# 1. Public Repository PR
python pr_review_agent.py https://github.com/microsoft/vscode/pull/100000

# 2. Another Public Repository
python pr_review_agent.py https://github.com/facebook/react/pull/25000

# 3. With GitHub Token (for private repos or higher rate limits)
# First, set your token:
$env:GITHUB_TOKEN="ghp_your_token_here"
# Then run:
python pr_review_agent.py https://github.com/your-org/your-repo/pull/123

# 4. Using token as command argument
python pr_review_agent.py https://github.com/your-org/your-repo/pull/123 --token ghp_your_token_here

# 5. Get help
python pr_review_agent.py --help

# To get a GitHub token:
# 1. Go to https://github.com/settings/tokens
# 2. Click "Generate new token (classic)"
# 3. Select scopes: "repo" for private repos, "public_repo" for public only
# 4. Generate and copy the token
# 5. Set it as environment variable or use --token flag
